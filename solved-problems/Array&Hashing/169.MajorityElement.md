class Solution:
    def majorityElement(self, nums: List[int]) -> int:
        map_freq={}
        n=len(nums)//2
        for num in nums:
            map_freq[num]=map_freq.get(num,0)+1
            if map_freq[num]>n:
                return num

class Solution:
    def majorityElement(self, nums: List[int]) -> int:
        king, supporters=None, 0
        for num in nums:
            if supporters==0:
                king=num
            if num==king:
                supporters+=1
            else:
                supporters-=1
        return king
